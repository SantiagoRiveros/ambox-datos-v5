CREATE TABLE clientes (
    id INTEGER PRIMARY KEY,
    nombre TEXT,
    ciudad TEXT
);

CREATE TABLE productos (
    id INTEGER PRIMARY KEY,
    nombre TEXT,
    precio REAL,
    categoria TEXT
);

CREATE TABLE ventas (
    id INTEGER PRIMARY KEY,
    cliente_id INTEGER,
    producto_id INTEGER,
    cantidad INTEGER,
    fecha TEXT,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (producto_id) REFERENCES productos(id)
);

Ejercicio 1 — Historial de ventas
Mostrar el nombre del cliente, el nombre del producto, la cantidad comprada y la fecha de cada venta.
Ejercicio 2 — Ventas de productos caros
Mostrar el nombre del cliente, el producto y su precio para todas las ventas de productos que cuesten más de $25.000.
Ejercicio 3 — Compras de una ciudad
Mostrar el nombre del cliente y el producto comprado por clientes que viven en La Plata.
Ejercicio 4 — Clientes sin compras
Mostrar el nombre y la ciudad de todos los clientes que todavía no realizaron ninguna compra. Utilizar LEFT JOIN.
Ejercicio 5 — Cantidad de unidades vendidas
